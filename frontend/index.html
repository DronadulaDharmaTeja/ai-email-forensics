const { onCall, HttpsError } = require("firebase-functions/v2/https"); const { Resend } = require("resend"); const resend = new Resend(process.env.RESEND_API_KEY); exports.sendWelcomeEmail = onCall( { region: "asia-south1", }, async (request) => { // Check that the Firebase user is authenticated if (!request.auth) { throw new HttpsError( "unauthenticated", "User must be authenticated." ); } // Get the email directly from Firebase Authentication const email = request.auth.token.email; if (!email) { throw new HttpsError( "invalid-argument", "Authenticated user does not have an email address." ); } try { const result = await resend.emails.send({ from: "ForensiMail ", to: [email], subject: "Welcome to ForensiMail", html: ` Welcome to ForensiMail
🛡️

Welcome to ForensiMail

AI Email Forensics Investigation System

Your ForensiMail account has been created successfully.

You can now sign in and access the AI Email Forensics Investigation System.

Account Email

${email}

Your account is ready. Please return to ForensiMail and sign in using your email address and password.

This is an automated email from ForensiMail. Please do not reply to this message.

`, }); console.log( "Welcome email sent successfully:", result ); return { success: true, email: email, message: "Welcome email sent successfully.", }; } catch (error) { console.error( "WELCOME EMAIL ERROR:", error ); throw new HttpsError( "internal", "Unable to send welcome email." ); } } );
